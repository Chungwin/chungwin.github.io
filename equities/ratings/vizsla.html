Why is there no cheap price?

A share counts as cheap here if, on average, it should be worth at least 1.72 times its price by Dec 2029. Worked out on paper, that would happen at US$1.87 a share. But at that price a share would still be worth US$1.89 if the project stalls, because that value is fixed at today's prices (what the market pays today for this company or for similar stalled projects) and does not fall with the share price. The sum would then say you make money even if everything goes wrong, which is not believable: if the share really fell that far, the value if it stalls would fall with it. So the page shows no cheap price instead of a misleading number.

The rating function itself (Python)The exact code behind the card, the sliders and this calculation
# Vizsla Silver (NYSE American: VZLA) - value rating v0.4.1 (16 Nov 2026). US$M; share values US$.
# Horizon: works date 31 Dec 2029 (3.13 years); lines carried to it (cheap 1.72x, expensive 1.077x).
# Rule set v0.4.1: a mine not yet built is valued by rule A1, never by the producer rule. A1 on EV / study NPV:
# silver 'funded / in construction' median (Silver Tiger 0.286, GoGold 0.598), with the build money that goes
# into the project (net initial cost 173 + working capital 25) and the cash left (92) added once, notes deducted.
# F3: q = 0.4; late = the works case without the 92 left (spent on holding costs) discounted 2 years at 10.668%.
# F1: stalled at AbraSilver US$181.86/oz AuEq (cap US$230.82), F2 raise.
DEFAULTS = dict(
    P=3.07,            # share price, US$ (start close 13 Nov 2026)
    pw=0.38, q=0.4, delay=2.0,
    ag=49.121, au=3886.3,
    ev_nav=[185.0 / 647.5, 684.0 / 1144.6],   # Silver Tiger, GoGold: EV / NPV at today's metal (table C-3(a) v2)
    build=173.0 + 25.0,  # US$M into the project (net initial cost, working capital; estimates)
    cash_left=92.0,      # US$M left at the works date (410.9 - 319 of spending; estimate)
    opt_money=36.1,      # US$M option money (works and late)
    notes=300.0,         # US$M convertible notes at face (conversion US$5.84, out of the money in every case below)
    stalled_per_oz=181.86, cap_per_oz=230.82, oz=4.242,
    cash_fail=255.1,     # US$M at end-2029 in a stall incl. 21.8 option money (estimate)
    years=3.132,
)
SH_WORKS = 382.16           # M shares incl. options, share units, shares to be issued
SH_FAIL0 = 378.20           # M before the F2 raise in a stall
R = 1.5 ** 0.25

def npv_today(ag, au):
    """Panuco FS after-tax NPV5, US$M: 1,802 at the deck (35.50 / 3,100), +668.5 per +25% (both metals),
    deck-revenue weights Ag 58.3% / Au 41.7%; straight line, last slope beyond +50%."""
    move = 0.583 * (ag / 35.50 - 1) + 0.417 * (au / 3100 - 1)
    return 1802 + move / 0.25 * ((3139 - 1802) / 2)

def _m(i):
    s = sorted(i['ev_nav']); return i.get('m', (s[0] + s[-1]) / 2 if len(s) == 2 else s[len(s) // 2])

def _branches(i, P):
    npv = npv_today(i['ag'], i['au'])
    Vw = _m(i) * npv + i['build'] + i['cash_left'] + i['opt_money'] - i['notes']
    Mw_ps = Vw / SH_WORKS
    # desk, 16 Nov (analyst v0.4.2): above US$5.84 a share the US$300M notes turn into 51.37M shares and are no longer deducted
    conv = Mw_ps > 5.84
    if conv:
        Mw_ps = (Vw + i['notes']) / (SH_WORKS + 51.37)
    Ml_ps = ((Vw - i['cash_left'] + (i['notes'] if conv else 0)) / (SH_WORKS + (51.37 if conv else 0))) / R ** i['delay']
    A = min(i['stalled_per_oz'], i['cap_per_oz']) * i['oz']
    s = max(0.0, i['notes'] - i['cash_fail'])
    if A - s <= 0: Ms_ps = 0.0
    else:
        p = 0.8 * (A - s) / SH_FAIL0; Ms_ps = A / (SH_FAIL0 + s / p)
    Mf_ps = i['q'] * Ml_ps + (1 - i['q']) * Ms_ps
    sh_f = SH_FAIL0 + (s / p if A - s > 0 else 0)
    return dict(npv=npv, m=_m(i), shares_works=SH_WORKS, shares_late=SH_WORKS, shares_fail=sh_f, Mw_ps=Mw_ps, M_late_ps=Ml_ps, M_stalled_ps=Ms_ps, Mf_ps=Mf_ps)

def value(inp=None):
    i = dict(DEFAULTS, **(inp or {})); P = i['P']; b = _branches(i, P)
    E = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P
    Eq0 = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['M_stalled_ps']) / P
    y = i['years']
    return dict(b, Mw=b['Mw_ps'] / P, Mf=b['Mf_ps'] / P, M_late=b['M_late_ps'] / P, M_stalled=b['M_stalled_ps'] / P,
                E=E, E_q0=Eq0, cheap_line=2.0 ** (y / 4), exp_line=1.1 ** (y / 4), per_year=E ** (1 / y) - 1)

def levels(inp=None):
    """No new shares in any branch and the notes stay out of the money: the value per share does not depend on
    the price, so the levels are V / line."""
    i = dict(DEFAULTS, **(inp or {})); r = value(i); V = r['E'] * i['P']
    cheap = V / r['cheap_line']; exp = V / r['exp_line']
    return dict(cheap_below=(cheap if cheap > r['M_stalled_ps'] else None), cheap_arith=cheap, expensive_above=exp,
                fail_ps=r['M_stalled_ps'], V=V)

if __name__ == '__main__':
    r = value(); l = levels()
    print({k: round(v, 4) for k, v in r.items()}); print(l)

DEFAULTS.update({'m': 0.8, 'pw': 0.35})
