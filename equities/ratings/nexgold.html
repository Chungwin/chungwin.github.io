# NexGold Mining (TSXV: NEXG) - value rating v0.4.1 (16 Nov 2026). C$M; share values C$.
# Rule set v0.4.1: A1 - the works state (financed, built, first gold about mid-2029) at the EV / study-NPV median
# of gold 'funded / in construction' (Denarius 0.535, Montage 0.885), with the money that goes into the project
# or stays as cash added once and the loan deducted; F3 - q = 0.4, the late build valued the same way with the
# extra spending funded by equity and discounted at the hurdle rate (10.668%/yr) for 1.5 years; E at q = 0 is the
# cheap test. F1 - stalled projects at the cap (NexGold's own US$23.9/oz on 5.986 Moz), F2 raise.
DEFAULTS = dict(
    P=1.12,            # share price, C$ (start close 13 Nov 2026)
    pw=0.50, q=0.4, delay=1.5,
    ev_nav=[173.1 / 323.8, 4275.6 / (0.9 * 4830.2)],   # Denarius, Montage: EV / NPV at today's metal (table C-3(a) v2)
    npv=2254.8,        # Goldboro after-tax NPV5 at today's gold, C$M (A2, X1; less Appian 2.9% royalty). Cost reset: 1953.9
    capex=369.9,       # C$M, C$271.0M (Dec 2021) escalated +36.48% (FRED PCU333131333131)
    reserve=37.0, other_uses=92.3,   # overrun reserve; G&A 19.6 + other projects 22.0 + Q4 spend 13.6 + interest 37.2 (estimates)
    debt=249.5,        # Appian US$175M, non-binding letter, counted as drawn in works and late
    goliath=99.6,      # Goliath at US$23.9/oz x 2.921 Moz
    late_extra=107.0,  # C$M extra spending that builds nothing in a late build (estimate)
    stalled=204.0,     # C$M stalled assets at the cap (US$23.9/oz x 5.986 Moz)
    fail_spend=96.55,  # C$M spending that builds nothing to 13 Nov 2030 (estimate)
    years=4.0,
)
CASH = 87.4
SHARES = 261.084264
UNITS = 3.010009 + 0.901507 + 0.568440
SPROTT_CAD = 3.375 * 1.5758 / 1.1052          # US$3.375M of royalty payments settled in shares at the price
R = 1.5 ** 0.25
SERIES = [(4.209523, 0.95), (1.580301, 1.00), (6.913623, 1.40), (69.425, 1.92), (12.5, 1.05), (1.386382, 0.84),
          (0.057501, 0.96), (0.059182, 4.64), (0.193481, 2.32), (1.166667, 1.00), (0.025, 0.77), (0.011941, 5.95),
          (0.01244, 4.02), (0.100263, 2.50), (0.6, 1.51), (0.15, 1.68), (0.217072, 0.73), (0.0775, 1.44),
          (0.155, 1.78), (0.15, 1.24)]       # warrants then options: (M, strike C$)

def npv_at(au):
    """Goldboro 2021 FS after-tax NPV5, C$M, at US$ gold: C$ price = US$ x 1.4258 (X1); C$328M at C$2,000, last
    slope C$0.57M per C$1; less Appian's 2.9% royalty as a 2.9% lower price."""
    c = au * 1.5758 / 1.1052
    return 328 + (c - 2000) * 0.57 - 0.029 * c * 0.57

def _m(i):
    s = sorted(i['ev_nav']); n = len(s)
    return i.get('m', s[n // 2] if n % 2 else (s[n // 2 - 1] + s[n // 2]) / 2)

def _npv(i):
    # desk, 16 Nov: a lower study value (the cost reset) is carried to other gold prices as the same C$ amount
    return i['npv'] if 'au' not in i else npv_at(i['au']) - (2254.8 - i['npv'])

def _built(i, P, extra):
    itm = [(n, k) for n, k in SERIES if k < P]
    rest = sorted([(n, k) for n, k in SERIES if k >= P], key=lambda x: x[1])
    cashP = sum(n * k for n, k in itm)
    gap = i['capex'] + i['reserve'] + i['other_uses'] - CASH - i['debt'] - cashP + extra
    raise_sh = max(gap, 0.0) / (0.8 * P); surplus = max(-gap, 0.0)
    sh = SHARES + UNITS + SPROTT_CAD / P + sum(n for n, _ in itm) + raise_sh
    V = _m(i) * _npv(i) + i['capex'] + i['reserve'] - i['debt'] + i['goliath'] + surplus
    for n, k in rest:                      # higher-priced series exercised when struck below the value per share
        if k < V / sh:
            V += n * k; sh += n
    return V / sh, sh, gap

def _stalled(i, P):
    ex = [(n, k) for n, k in SERIES[:1] if k < P]          # only the C$0.95 series expiring 11 Dec 2026
    n0 = SHARES + UNITS + SPROTT_CAD / P + sum(n for n, _ in ex)
    funds = CASH + sum(n * k for n, k in ex) + 4.0 - i['fail_spend']
    s = max(-funds, 0.0); A = i['stalled'] + max(funds, 0.0)
    if A - s <= 0: return 0.0, n0
    p = 0.8 * (A - s) / n0
    return A / (n0 + s / p), n0 + s / p

def _branches(i, P):
    Mw_ps, sh_w, gap = _built(i, P, 0.0)
    Ml0, sh_l, _ = _built(i, P, i['late_extra'])
    Ml_ps = Ml0 / R ** i['delay']
    Ms_ps, sh_f = _stalled(i, P)
    Mf_ps = i['q'] * Ml_ps + (1 - i['q']) * Ms_ps
    return dict(m=_m(i), gap=gap, shares_works=sh_w, shares_late=sh_l, shares_fail=sh_f, Mw_ps=Mw_ps,
                M_late_ps=Ml_ps, M_stalled_ps=Ms_ps, Mf_ps=Mf_ps)

def value(inp=None):
    i = dict(DEFAULTS, **(inp or {})); P = i['P']; b = _branches(i, P)
    E = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P
    Eq0 = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['M_stalled_ps']) / P
    return dict(b, Mw=b['Mw_ps'] / P, Mf=b['Mf_ps'] / P, M_late=b['M_late_ps'] / P, M_stalled=b['M_stalled_ps'] / P,
                E=E, E_q0=Eq0, per_year=E ** (1 / i['years']) - 1)

def _E(i, P):
    b = _branches(i, P); return (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P, b

def _cross(i, line, hi, lo=0.02, steps=4000):
    import math
    prev = hi
    for k in range(steps + 1):
        P = hi * math.exp(math.log(lo / hi) * k / steps)
        if _E(i, P)[0] >= line:
            a, b = P, prev
            for _ in range(60):
                mid = (a + b) / 2
                a, b = (mid, b) if _E(i, mid)[0] >= line else (a, mid)
            return a
        prev = P
    return None

def levels(inp=None):
    i = dict(DEFAULTS, **(inp or {})); y = i['years']; hi = 10 * i['P']
    exp = _cross(i, 1.1 ** (y / 4), hi); cheap = _cross(i, 2.0 ** (y / 4), hi)
    fps = _E(i, cheap)[1]['M_stalled_ps'] if cheap else None
    return dict(expensive_above=exp, cheap_below=(cheap if cheap and cheap > fps else None),
                cheap_arith=cheap, fail_ps_at_cheap=fps)
