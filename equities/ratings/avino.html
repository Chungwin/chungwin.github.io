Why is there no cheap price?

A share counts as cheap here if, on average, it should be worth at least 2.00 times its price by Nov 2030. Worked out on paper, that would happen at US$2.38 a share. But at that price a share would still be worth US$4.36 if the project stalls, because that value is fixed at today's prices (what the market pays today for this company or for similar stalled projects) and does not fall with the share price. The sum would then say you make money even if everything goes wrong, which is not believable: if the share really fell that far, the value if it stalls would fall with it. So the page shows no cheap price instead of a misleading number.

Why

At US$4.29 the expected value in four years is 1.29 times today's price, about 6.5% a year. The Avino mine is valued like the three silver-led producers (14.85 times its US$37.6M yearly margin at today's metal). La Preciosa has no study, so a finished mill is valued at what the market pays per ounce for silver mines being built (US$283.69 per gold-equivalent ounce) plus its estimated US$250M build cost. If the mill is built by 2030 (the desk's chance: 15%) a share is worth 1.85 times today's price; if not, 1.19 times (a late mill is 3 in 10 of these cases). At the median multiple of all producers (6.36×) the same sum reads 1.01×, expensive.

What would change the rating

Price alone won't make this cheap. A La Preciosa study with its build cost, recovery and a date would be the first checkable reason in either direction.

A price above US$5.08 with nothing else changed; the mine valued at less than 8.4 times its margin; silver and gold both about 10% lower (expected value 1.05×); or a Q4 2026 all-in sustaining cost (AISC) above US$32 per silver-equivalent ounce at guidance prices.

# Avino Silver & Gold Mines (NYSE American: ASM) - value rating v0.4.1 (16 Nov 2026). US$M; share values US$.
# Rule set v0.4.1: P1 - the producing Avino mine at the silver-led producer median 14.85x its AISC margin at
# today's metal (low 6.36x, high 14.935x); a mine not yet built is never valued by the producer rule, so a built
# La Preciosa (no study, no NPV) takes A2's fallback: table C-3(a) v2 EV per oz AuEq of silver 'funded / in
# construction' (US$283.69), with the money that goes into the project added once. F3: q = 0.3, late = works
# value discounted at the hurdle rate (10.668%/yr) for 1.5 years. F1: stalled La Preciosa at the silver
# resource-stage median (US$60.68/oz AuEq) capped at its own value per oz at the start close.
DEFAULTS = dict(
    P=4.29,            # share price, US$ (start close 13 Nov 2026)
    pw=0.15,           # chance it works (desk)
    ag=49.121, au=3886.3,
    mult=14.85,        # Avino mine EV / AISC margin
    lp_per_oz=283.69,  # built La Preciosa, US$ per oz AuEq (silver funded / in construction median)
    lp_oz=2.799,       # Moz AuEq, La Preciosa resource (all categories)
    stalled_per_oz=60.68,
    q=0.3, delay=1.5,
    capex=250.0,       # US$M, estimate (no filing figure)
    studies=15.0, work_cash=25.0,
    elasticity=1.51,   # works value per oz moves 1.51x the metal move (median of the desk's re-based silver studies)
    years=4.0,
)
SHARES = 169.368596
RSUS = 1.825166
CAD = 1.1052 / 1.5758                          # US$ per C$
OPTIONS = [(0.450, 1.12), (0.535, 0.78), (1.494, 2.11), (0.075, 4.38), (0.436572, 9.41), (0.300, 9.47)]  # (M, C$)
CASH, LOANS = 136.9, 1.7                        # usable cash; equipment loans (30 Sep 2026)
EV_TODAY = 590.1
R = (1.5 ** 0.25)                               # hurdle rate per year, 10.668%

def mine_margin(ag, au):
    return 53.43 * ag / 49.121 + 29.22 * au / 3886.3 + 26.39 - 71.41   # revenue (Ag, Au, Cu fixed) - AISC

def lp_revenue(ag, au):
    return 209.86 * ag / 49.121 + 25.14 * au / 3886.3

def fcf4(ag, au):
    """4-year free cash flow at the metal price: US$43.8M at today's metal, moving with the mine's revenue less 30.5% tax."""
    rev = 53.43 * ag / 49.121 + 29.22 * au / 3886.3 + 26.39
    return 43.8 + 4 * (rev - 109.04) * (1 - 0.305)

def _branches(i, P):
    ag, au = i['ag'], i['au']
    itm = [(n, k) for n, k in OPTIONS if k * CAD < P]
    n_opt = sum(n for n, _ in itm); money = sum(n * k for n, k in itm) * CAD
    mine = i['mult'] * mine_margin(ag, au)
    f4 = fcf4(ag, au)
    move = lp_revenue(ag, au) / lp_revenue(49.121, 3886.3) - 1
    if i.get('lp_mode', 'A1') == 'A1':
        lp_built = i['lp_per_oz'] * i['lp_oz'] * (1 + i['elasticity'] * move)
        capex_credit = i['capex']                  # money into the project, added once (EV-per-unit yardstick)
    else:                                          # v0.4 reading (producer rule on the unbuilt mine), for checks only
        lp_built = i['lp_mult'] * (lp_revenue(ag, au) - 104.65)
        capex_credit = 0.0
    gap = i['capex'] + i['studies'] + i['work_cash'] - CASH - f4
    new = max(0.0, gap) / (0.8 * P)
    extra_cash = max(0.0, -gap)
    n0 = SHARES + RSUS + n_opt
    Vw = mine + lp_built + capex_credit + i['work_cash'] + extra_cash - LOANS + money
    Mw_ps = Vw / (n0 + new)
    Ml_ps = Mw_ps / R ** i['delay']
    cap = (EV_TODAY - i['mult'] * mine_margin(49.121, 3886.3)) / i['lp_oz']    # F1 cap at the start close
    st = min(i['stalled_per_oz'], max(cap, 0.0)) * i['lp_oz']
    Vs = mine + st + CASH + f4 - i['studies'] - LOANS + money
    Ms_ps = Vs / n0
    Mf_ps = i['q'] * Ml_ps + (1 - i['q']) * Ms_ps
    return dict(mine=mine, lp_built=lp_built, fcf4=f4, gap=gap, new_shares=new, shares_works=n0 + new,
                shares_fail=n0, Mw_ps=Mw_ps, M_late_ps=Ml_ps, M_stalled_ps=Ms_ps, Mf_ps=Mf_ps, cap_per_oz=cap)

def value(inp=None):
    i = dict(DEFAULTS, **(inp or {})); P = i['P']; b = _branches(i, P)
    E = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P
    Eq0 = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['M_stalled_ps']) / P
    return dict(b, Mw=b['Mw_ps'] / P, Mf=b['Mf_ps'] / P, M_late=b['M_late_ps'] / P, M_stalled=b['M_stalled_ps'] / P,
                E=E, E_q0=Eq0, per_year=E ** (1 / i['years']) - 1)

def _E(i, P):
    b = _branches(i, P); return (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P, b

def _cross(i, line, hi, lo=0.02, steps=4000):
    import math
    prev = hi
    for k in range(steps + 1):
        P = hi * math.exp(math.log(lo / hi) * k / steps)
        if _E(i, P)[0] >= line:
            a, b = P, prev
            for _ in range(60):
                mid = (a + b) / 2
                a, b = (mid, b) if _E(i, mid)[0] >= line else (a, mid)
            return a
        prev = P
    return None

def levels(inp=None):
    i = dict(DEFAULTS, **(inp or {})); y = i['years']; hi = 10 * i['P']
    exp = _cross(i, 1.1 ** (y / 4), hi); cheap = _cross(i, 2.0 ** (y / 4), hi)
    fps = _E(i, cheap)[1]['M_stalled_ps'] if cheap else None
    return dict(expensive_above=exp, cheap_below=(cheap if cheap and cheap > fps else None),
                cheap_arith=cheap, fail_ps_at_cheap=fps)

if __name__ == '__main__':
    r = value(); l = levels()
    print({k: round(v, 4) for k, v in r.items()}); print(l)
