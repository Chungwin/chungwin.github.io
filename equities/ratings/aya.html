Why is there no cheap price?

A share counts as cheap here if, on average, it should be worth at least 2.00 times its price by Nov 2030. Worked out on paper, that would happen at C$17.46 a share. But at that price a share would still be worth C$29.36 if the project stalls, because that value is fixed at today's prices (what the market pays today for this company or for similar stalled projects) and does not fall with the share price. The sum would then say you make money even if everything goes wrong, which is not believable: if the share really fell that far, the value if it stalls would fall with it. So the page shows no cheap price instead of a misleading number.

The rating function itself (Python)The exact code behind the card, the sliders and this calculation
# Aya Gold & Silver (TSX: AYA) - value rating v0.4.1 (16 Nov 2026). US$M unless marked; share values in C$.
# Rule set v0.4.1: P1 - producing Zgounder at the silver-led producer median 14.85x its AISC margin at today's
# metal (low 6.36x, high 14.935x); A1 - Boumadine (works state: a funded build) at the EV / study-NPV median of
# gold 'funded / in construction' (Denarius; Montage on its 90% of the NPV), with the money that goes into the project added once;
# F3 - q = 0 (the works state is not production); F1 - stalled Boumadine at the gold PEA median per oz.
USD_PER_CAD = 1.1052 / 1.5758                 # US$ per C$, ECB 13 Nov 2026
DEFAULTS = dict(
    P=32.58,           # share price, C$ (start close 13 Nov 2026)
    pw=0.39,           # chance it works (desk)
    ag=49.121, au=3886.3,  # COMEX closes 13 Nov 2026, US$/oz
    mult=14.85,        # Zgounder EV / AISC margin (silver-led producers' median, table C-3(a) v2)
    ev_nav=[173.1 / 323.8, 4275.6 / (0.9 * 4830.2)],  # gold funded / in construction, EV / study NPV today (Denarius, Montage)
    q=0.0,             # share of misses that are late builds (rule F3: 0, works state is a funded build)
    zg_oz=5.7,         # Moz silver a year, 2026 guidance midpoint
    cash_cost=17.53,   # US$/oz, Q3 2026
    sust=4 * (5.929 + 6.790),  # US$M a year of capital spending + G&A in the AISC estimate (AISC est. US$26.46/oz)
    capex=463.0, studies=85.0, wasted_fail=130.0,
    stalled_oz=4.5645, stalled_per_oz=188.53,  # Moz AuEq attributable; US$/oz gold PEA median
    years=4.0,
)
SHARES = 144.195352
OPTIONS = [(2.890944, 1.43), (0.314667, 4.75), (0.063300, 7.69), (4.870000, 15.63), (0.500000, 14.45)]  # (M, C$)
UNITS = 0.929067 + 0.080174 + 0.244973
NET_CASH = 183.791 + 16.405 - 71.429           # 30 Sep 2026
EV_TODAY = 3182.7                              # US$M at the start close
B_ROWS = [(5441, 0), (8169, 1808), (10995, 3537), (14237, 5525), (16246, 6738)]  # PEA: LOM revenue -> NPV5

def boumadine_npv(au, ag):
    rev = 10995 + 1.84828 * (au - 3500) + 68.99195 * (ag - 50)   # payable Au 1,848.28 koz, Ag 68.99 Moz
    for (r0, v0), (r1, v1) in zip(B_ROWS, B_ROWS[1:]):
        if rev <= r1 or (r1, v1) == B_ROWS[-1]:
            return v0 + (rev - r0) * (v1 - v0) / (r1 - r0)

def _parts(i):
    margin = i['zg_oz'] * (i['ag'] - i['cash_cost']) - i['sust']
    zg = i['mult'] * margin
    profit = 4 * (margin + 4 * 5.929 - 31.0) - 7.337
    fcf4 = 4 * margin - 0.368 * profit - 7.337
    npv_b = 0.85 * boumadine_npv(i['au'], i['ag'])
    s = sorted(i['ev_nav']); m = (s[0] + s[-1]) / 2 if len(s) == 2 else s[len(s) // 2]
    m = i.get('m', m)
    return margin, zg, fcf4, npv_b, m

def _branches(i, P):
    margin, zg, fcf4, npv_b, m = _parts(i)
    itm = [(n, k) for n, k in OPTIONS if k < P]
    n0 = SHARES + UNITS + sum(n for n, _ in itm)
    opt_cash = sum(n * k for n, k in itm) * USD_PER_CAD
    gap = max(0.0, i['capex'] + i['studies'] - NET_CASH - fcf4)
    new = gap / (0.8 * P * USD_PER_CAD)
    surplus = max(0.0, NET_CASH + fcf4 - i['capex'] - i['studies'])
    Vw = zg + m * npv_b + i['capex'] + surplus + opt_cash              # money into the project added once
    Mw_ps = Vw / (n0 + new) / USD_PER_CAD
    margin0 = i['zg_oz'] * (49.121 - i['cash_cost']) - i['sust']
    cap = (EV_TODAY - i['mult'] * margin0) / i['stalled_oz']            # F1 cap, set at the start close (today's metal)
    st = min(i['stalled_per_oz'], max(cap, 0.0)) * i['stalled_oz']
    Vs = zg + NET_CASH + fcf4 - i['wasted_fail'] + st + opt_cash
    Ms_ps = Vs / n0 / USD_PER_CAD
    Ml_ps = Ms_ps   # q = 0 by rule F3 here; a late build would need its own value
    Mf_ps = i['q'] * Ml_ps + (1 - i['q']) * Ms_ps
    return dict(margin=margin, zgounder=zg, fcf4=fcf4, npv_b=npv_b, m=m, gap=gap, new_shares=new,
                shares_works=n0 + new, shares_fail=n0, Mw_ps=Mw_ps, Mf_ps=Mf_ps, M_stalled_ps=Ms_ps, cap_per_oz=cap)

def value(inp=None):
    i = dict(DEFAULTS, **(inp or {})); P = i['P']; b = _branches(i, P)
    E = (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P
    return dict(b, Mw=b['Mw_ps'] / P, Mf=b['Mf_ps'] / P, E=E, per_year=E ** (1 / i['years']) - 1)

def _E(i, P):
    b = _branches(i, P); return (i['pw'] * b['Mw_ps'] + (1 - i['pw']) * b['Mf_ps']) / P, b

def _cross(i, line, hi, lo=0.05, steps=4000):
    import math
    prev = hi
    for k in range(steps + 1):
        P = hi * math.exp(math.log(lo / hi) * k / steps)
        if _E(i, P)[0] >= line:
            a, b = P, prev
            for _ in range(60):
                mid = (a + b) / 2
                a, b = (mid, b) if _E(i, mid)[0] >= line else (a, mid)
            return a
        prev = P
    return None

def levels(inp=None):
    i = dict(DEFAULTS, **(inp or {})); y = i['years']; hi = 10 * i['P']
    exp = _cross(i, 1.1 ** (y / 4), hi); cheap = _cross(i, 2.0 ** (y / 4), hi)
    fps = _E(i, cheap)[1]['M_stalled_ps'] if cheap else None
    return dict(expensive_above=exp, cheap_below=(cheap if cheap and cheap > fps else None),
                cheap_arith=cheap, fail_ps_at_cheap=fps)

    r = value(); l = levels()
    print({k: round(v, 4) for k, v in r.items()}); print(l)
